import assert from "node:assert/strict";
import { describe, it } from "node:test";
import { network } from "hardhat";
import { encodeAbiParameters, keccak256, parseAbiParameters, toHex } from "viem";
import { computeClearingPrice } from "../../apps/api/src/clearing.js";

const WAD = 10n ** 18n;
const FLOOR = 10n ** 15n;

function rng(seed: number) {
  let state = seed >>> 0;
  return () => ((state = (state * 1664525 + 1013904223) >>> 0) / 2 ** 32);
}

describe("settlement fuzz", () => {
  it("accepts computed price, rejects alternatives, caps allocations, and releases ETH", async () => {
    for (let seed = 1; seed <= 12; seed++) {
      const random = rng(seed);
      const { viem, provider } = await network.connect();
      const publicClient = await viem.getPublicClient();
      const wallets = await viem.getWalletClients();
      const owner = wallets[0]!;
      const bidders = wallets.slice(1, 1 + Math.min(9, wallets.length - 1));
      const count = 2 + Math.floor(random() * bidders.length);
      const bids = bidders.slice(0, count).map(() => ({
        price: FLOOR * BigInt(1 + Math.floor(random() * 5)),
        amount: WAD * BigInt(1 + Math.floor(random() * 20)),
      }));
      const total = bids.reduce((sum, bid) => sum + bid.amount, 0n);
      const supply = WAD * BigInt(1 + Math.floor(random() * Number(total / WAD + 10n)));
      const registry = await viem.deployContract("AgentRegistry", [1n, total + WAD, 1n]);
      const auction = await viem.deployContract("LaunchpadAuction", [registry.address]);
      const token = await viem.deployContract("LaunchToken", ["Fuzz", "FUZ", supply, owner.account.address]);
      await token.write.approve([auction.address, supply]);
      const now = (await publicClient.getBlock()).timestamp;
      await auction.write.createAuction([token.address, supply, FLOOR, now + 100n, now + 1_000n, now + 2_000n]);
      for (const bidder of bidders.slice(0, count)) await registry.write.register(["0x"], { account: bidder.account, value: 1n });
      await provider.request({ method: "evm_setNextBlockTimestamp", params: [Number(now + 100n)] }); await provider.request({ method: "evm_mine", params: [] });
      const salts = bids.map((_, i) => toHex(BigInt(seed * 100 + i), { size: 32 }));
      for (let i = 0; i < count; i++) {
        const bidder = bidders[i]!; const bid = bids[i]!;
        const commitment = keccak256(encodeAbiParameters(parseAbiParameters("uint256,address,uint256,uint256,bytes32"), [0n, bidder.account.address, bid.price, bid.amount, salts[i]!]));
        await auction.write.commit([0n, commitment], { account: bidder.account, value: bid.price * bid.amount / WAD });
      }
      await provider.request({ method: "evm_setNextBlockTimestamp", params: [Number(now + 1_000n)] }); await provider.request({ method: "evm_mine", params: [] });
      for (let i = 0; i < count; i++) await auction.write.reveal([0n, bids[i]!.price, bids[i]!.amount, salts[i]!], { account: bidders[i]!.account });
      await provider.request({ method: "evm_setNextBlockTimestamp", params: [Number(now + 2_000n)] }); await provider.request({ method: "evm_mine", params: [] });
      const clearing = computeClearingPrice(bids, supply, FLOOR);
      const alternatives = new Set([FLOOR, ...bids.map((bid) => bid.price)]);
      alternatives.delete(clearing);
      for (const alternative of alternatives) {
        await viem.assertions.revertWithCustomError(auction.write.settle([0n, alternative]), auction, "InvalidClearingPrice");
      }
      await auction.write.settle([0n, clearing]);
      let allocated = 0n;
      for (let i = 0; i < count; i++) {
        allocated += await auction.read.allocationOf([0n, bidders[i]!.account.address]);
        await auction.write.claim([0n], { account: bidders[i]!.account });
      }
      assert(allocated <= supply, `seed ${seed}: allocation exceeds supply`);
      await auction.write.withdrawProceeds([0n]);
      assert.equal(await publicClient.getBalance({ address: auction.address }), 0n, `seed ${seed}: trapped ETH`);
    }
  });
});
